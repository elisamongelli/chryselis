Table ACT_ANAGRAFICA_PIANTE_TESTATA {

  DATA_INSERIMENTO TIMESTAMP
  DATA_ULTIMA_MODIFICA TIMESTAMP
}

  ID_PIANTA VARCHAR2(100) [primary key]
  NOME_PIANTA VARCHAR2(500)
  DESCRIZIONE_PIANTA CLOB
  FOTO_PIANTA BLOB
  ID_STANZA VARCHAR2(100)
  POSIZIONE_STANZA_X INTEGER
  POSIZIONE_STANZA_Y INTEGER
}

  ID_PIANTA VARCHAR2(100)
  UMIDITA_CORRENTE INTEGER
  ACQUA_ULTIMA_INNAFFIATURA INTEGER
  ALTRO_DATO_SENSORI_1 DECIMAL(5,2)
  ALTRO_DATO_SENSORI_2 DECIMAL(5,2)
  ALTRO_DATO_SENSORI_3 DECIMAL(5,2)
  ALTRO_DATO_SENSORI_4 DECIMAL(5,2)
}

Table HIS_ANAGRAFICA_PIANTE_DETTAGLIO_SENSORI {
  ID_PIANTA VARCHAR2(100)
  ID_STATO_PIANTA VARCHAR2(500)
  ID_PROGRAMMA_ESEGUITO VARCHAR2(100)
  UMIDITA_CORRENTE INTEGER
  ACQUA_ULTIMA_INNAFFIATURA INTEGER
  ALTRO_DATO_SENSORI_1 DECIMAL(5,2)
  ALTRO_DATO_SENSORI_2 DECIMAL(5,2)
  ALTRO_DATO_SENSORI_3 DECIMAL(5,2)
  ALTRO_DATO_SENSORI_4 DECIMAL(5,2)
  DATA_MODIFICA TIMESTAMP
}

Table LKP_PROGRAMMI {
  ID_PROGRAMMA VARCHAR2(100) [primary key]
  VALORE_SEQUENZA_TEMPORALE_PROGRAMMA INTEGER
  NOME_PROGRAMMA VARCHAR2(500)
  ORARIO_INIZIO_PROGRAMMA TIME
  ORARIO_FINE_PROGRAMMA TIME
}

Table LKP_PIANTE_PROGRAMMI {
  ID_PIANTA VARCHAR2(100)
  ID_PROGRAMMA VARCHAR2(100)
}

Table LKP_STATI_PIANTE {
  ID_STATO LISTA_STATI [primary key]
  NOME_STATO VARCHAR2(500)
  DESCRIZIONE_STATO CLOB
}

Table LKP_STANZE {
  ID_STANZA VARCHAR2(100) [primary key]
  NOME_STANZA VARCHAR2(500)
  DIMENSIONE_GRIGLIA_X INTEGER
  DIMENSIONE_GRIGLIA_Y INTEGER
}
